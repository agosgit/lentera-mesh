// ================= AIS ENCODE =================
struct AISMessage18 {
  uint8_t message_id = 18;      // 6 bits
  uint8_t repeat_indicator = 0; // 2 bits
  uint32_t mmsi;                // 30 bits
  uint8_t reserved = 0;         // 8 bits
  uint16_t sog;                 // 10 bits (Speed Over Ground * 10)
  bool position_accuracy = false; // 1 bit
  float longitude;              // 28 bits (1/10000 minute = 1/600000 derajat)
  float latitude;               // 27 bits
  uint16_t cog;                 // 12 bits (Course Over Ground * 10)
  uint16_t heading;             // 9 bits (True Heading)
  uint8_t utc_second;           // 6 bits
  uint8_t regional = 0;         // 2 bits
  bool cs_flag = false;         // 1 bit
  bool display_flag = false;    // 1 bit
  bool dsc_flag = false;        // 1 bit
  bool band_flag = false;       // 1 bit
  bool msg22_flag = false;      // 1 bit
  bool assigned_flag = false;   // 1 bit
  bool raim_flag = false;       // 1 bit
  uint32_t radio_status = 0;    // 20 bits
};

void writeBits(uint8_t *buffer, uint32_t value, uint16_t startBit, uint8_t numBits, bool isSigned = false)
{
  if (isSigned && (value & (1UL << (numBits - 1))))
    value |= 0xFFFFFFFFUL << numBits;
  for (uint8_t i = 0; i < numBits; i++)
  {
    uint16_t pos = startBit + i;
    uint8_t byteIdx = pos / 8;
    uint8_t bitIdx = 7 - (pos % 8);
    if (value & (1UL << (numBits - 1 - i)))
      buffer[byteIdx] |= (1 << bitIdx);
    else
      buffer[byteIdx] &= ~(1 << bitIdx);
  }
}

void bitsToAISString(const uint8_t *bits, uint16_t bitCount, char *output)
{
  uint16_t charCount = (bitCount + 5) / 6;
  for (uint16_t i = 0; i < charCount; i++)
  {
    uint8_t sixBits = 0;
    for (uint8_t j = 0; j < 6; j++)
    {
      uint16_t bitIdx = i * 6 + j;
      if (bitIdx < bitCount)
        sixBits = (sixBits << 1) | ((bits[bitIdx / 8] >> (7 - (bitIdx % 8))) & 1);
      else
        sixBits <<= 1;
    }
    output[i] = (sixBits < 40) ? (sixBits + 48) : (sixBits + 56);
  }
  output[charCount] = '\0';
}

String encodeMessage18(const AISMessage18 &msg) {
  uint8_t bits[21] = {0};  // 168 bits = 21 bytes

  writeBits(bits, msg.message_id, 0, 6);
  writeBits(bits, msg.repeat_indicator, 6, 2);
  writeBits(bits, msg.mmsi, 8, 30);
  writeBits(bits, msg.reserved, 38, 8);
  writeBits(bits, msg.sog, 46, 10);
  writeBits(bits, msg.position_accuracy ? 1 : 0, 56, 1);
  int32_t lng_enc = (int32_t)round(msg.longitude * 600000.0);
  writeBits(bits, lng_enc, 57, 28, true);
  int32_t lat_enc = (int32_t)round(msg.latitude * 600000.0);
  writeBits(bits, lat_enc, 85, 27, true);
  writeBits(bits, msg.cog, 112, 12);
  writeBits(bits, msg.heading, 124, 9);
  writeBits(bits, msg.utc_second, 133, 6);
  writeBits(bits, msg.regional, 139, 2);
  writeBits(bits, msg.cs_flag ? 1 : 0, 141, 1);
  writeBits(bits, msg.display_flag ? 1 : 0, 142, 1);
  writeBits(bits, msg.dsc_flag ? 1 : 0, 143, 1);
  writeBits(bits, msg.band_flag ? 1 : 0, 144, 1);
  writeBits(bits, msg.msg22_flag ? 1 : 0, 145, 1);
  writeBits(bits, msg.assigned_flag ? 1 : 0, 146, 1);
  writeBits(bits, msg.raim_flag ? 1 : 0, 147, 1);
  writeBits(bits, msg.radio_status, 148, 20);

  char payload[29];
  bitsToAISString(bits, 168, payload);
  return String(payload);
}

String nmeaChecksum(const String &sentence)
{
  int chk = 0;
  for (int i = 1; i < sentence.length(); i++)
    chk ^= sentence[i];
  char buf[5];
  sprintf(buf, "*%02X", chk);
  return String(buf);
}
String makeAivdm(const String &payload)
{
  String base = "!AIVDM,1,1,,A," + payload + ",0";
  return base + nmeaChecksum(base);
}

void loadConfig() {

  if (!f) {
    Serial.println("config.json belum ada, pakai default!");
    return;
  }
  StaticJsonDocument<256> doc;
  DeserializationError err = deserializeJson(doc, f);
  if (err) {
    Serial.println("Gagal baca config.json! Pakai default.");
    f.close();
    return;
  }
  if (doc.containsKey("mmsi")) config.mmsi = String((const char*)doc["mmsi"]);
  if (doc.containsKey("ship")) config.shipName = String((const char*)doc["ship"]);
  if (doc.containsKey("apn")) config.apn = String((const char*)doc["apn"]);
  if (doc.containsKey("simUser")) config.simUser = String((const char*)doc["simUser"]);
  if (doc.containsKey("simPass")) config.simPass = String((const char*)doc["simPass"]);
  if (doc.containsKey("koneksi")) config.koneksi = String((const char*)doc["koneksi"]);
  if (doc.containsKey("wifiSSID")) config.wifiSSID = String((const char*)doc["wifiSSID"]);
  if (doc.containsKey("wifiPass")) config.wifiPass = String((const char*)doc["wifiPass"]);
  if (doc.containsKey("mqttHost")) config.mqttHost = String((const char*)doc["mqttHost"]);
  if (doc.containsKey("mqttPort")) config.mqttPort = doc["mqttPort"].as<int>();
  if (doc.containsKey("mqttUser")) config.mqttUser = String((const char*)doc["mqttUser"]);
  if (doc.containsKey("mqttPass")) config.mqttPass = String((const char*)doc["mqttPass"]);
  if (doc.containsKey("mqttTopic")) config.mqttTopic = String((const char*)doc["mqttTopic"]);

  Serial.printf("Config loaded dari SPIFFS: MMSI=%s, Ship=%s\n", config.mmsi.c_str(), config.shipName.c_str());
  f.close();
}

void handleUpdateUpload()
{
  HTTPUpload &upload = server.upload();
  static bool started = false;
  if (upload.status == UPLOAD_FILE_START) {
    started = Update.begin(UPDATE_SIZE_UNKNOWN);
    Serial.println("[OTA] Mulai upload firmware...");
  } else if (upload.status == UPLOAD_FILE_WRITE) {
    if (started) Update.write(upload.buf, upload.currentSize);
  } else if (upload.status == UPLOAD_FILE_END) {
    if (Update.end(true)) {
      Serial.printf("[OTA] Selesai upload: %u bytes\n", upload.totalSize);
    } else {
      Serial.println("[OTA] Gagal update!");
    }
  }
}

// ======= Filter Duplikat AIS Super Ketat =======
String lastPayload = "";
unsigned long lastPayloadTime = 0;
const unsigned long DUP_WINDOW = 7000; // 7 detik jendela duplikat

bool isDuplicate(String data) {
  // Bersihkan string dari karakter tersembunyi
  data.trim(); 
  data.replace("\r", "");
  data.replace("\n", "");

  // Ambil payload (setelah tanda ':' terakhir)
  int lastColon = data.lastIndexOf(':');
  if (lastColon < 0) return false;
  String payload = data.substring(lastColon + 1);
  payload.trim();
  payload.replace("\r", "");
  payload.replace("\n", "");

  // Jika payload identik & masih dalam window waktu
  if (payload == lastPayload && (millis() - lastPayloadTime) < DUP_WINDOW) {
    Serial.printf("⚠️ Duplikat AIS diabaikan (Δt=%lu ms)\n", millis() - lastPayloadTime);
    return true;
  }

  // Simpan payload baru
  lastPayload = payload;
  lastPayloadTime = millis();
  return false;
}


// ================= HANDLERS =================
void gw_handlePacket(String data, int rssi) {
  if (data.startsWith("PINK:")) {
    String chain = data.substring(5);
    String temp = chain; temp.replace("-", "");
    int hopCount = chain.length() - temp.length();
    Serial.printf("📶 Mesh Route (%d hop): %s (RSSI %d)\n", hopCount, chain.c_str(), rssi);
  }

  else if (data.startsWith("MSG:")) {
    if (isDuplicate(data)) {
      Serial.println("⚠️ Duplikat AIS diabaikan");
      return;
    }

    int colonIdx = data.indexOf(':', 4);
    String route = (colonIdx > 0) ? data.substring(4, colonIdx) : "";
    String payload = (colonIdx > 0) ? data.substring(colonIdx + 1) : data.substring(4);

    String nmea = makeAivdm(payload);
    Serial.printf("📩 AIS via [%s] RSSI:%d\n", route.c_str(), rssi);

    if (mqttClient.connected()) {
      mqttClient.publish(config.mqttTopic.c_str(), nmea.c_str());
      Serial.printf("📡 MQTT Publish: %s\n", nmea.c_str());
    }
  }




}

void mqttReconnect() {
  if (mqttClient.connected()) return;

  if (millis() - lastMqttReconnect < 5000) return;
  lastMqttReconnect = millis();

  Serial.print("🔌 MQTT connecting...");

  if (mqttClient.connect("ESP32GSM", config.mqttUser.c_str(), config.mqttPass.c_str())) {
    Serial.println(" connected!");
    // digitalWrite(LED_RUN, HIGH);   // LED nyala kalau berhasil connect MQTT
  } else {
    Serial.printf(" failed rc=%d\n", mqttClient.state());
  }
}

// ================== SETUP GW ==================
void setup() {
  Serial.begin(115200);
  pinMode(LED_TX, OUTPUT);
  pinMode(LED_RX, OUTPUT);
  pinMode(LED_SETTING, OUTPUT);
  pinMode(LED_RUN, OUTPUT);
  pinMode(BTN_MODE, INPUT_PULLUP);
  digitalWrite(LED_SETTING, HIGH); 
  digitalWrite(LED_RUN, HIGH); 
  delay(2000); 
  // digitalWrite(LED_SETTING, LOW);
  digitalWrite(LED_RUN, LOW);
  digitalWrite(LED_SETTING, LOW);

  // ---- Inisialisasi SPIFFS ----
  MAC_FULL = ESP.getEfuseMac();
  // NODE_ID = (uint32_t)(MAC_FULL & 0xFFFF);
  if (!SPIFFS.begin(true)) {
    Serial.println("SPIFFS mount gagal!");
  } else {
    Serial.println("SPIFFS ready");
    loadConfig();
  }

  // ---- Mode konfigurasi (jika tombol ditekan) ----
  unsigned long t0 = millis();
  bool hold = false;
  while (millis() - t0 < HOLD_MS) {
    if (digitalRead(BTN_MODE) == LOW) hold = true;
    else { hold = false; break; }
  }
  if (hold) {
    WiFi.mode(WIFI_AP);
    WiFi.softAP("ESP32-OTA-GATE", "mesh12345");
    server.on("/", HTTP_GET, sendRoot);
    server.on("/update", HTTP_POST, handleUpdateFinish, handleUpdateUpload);
    server.on("/config", HTTP_POST, handleConfigPost);
    server.begin();
    Serial.println("⚙️ Mode Setting aktif (AP: ESP32-OTA-GATE)");
    while (true) {
      server.handleClient();
      // LED_SETTING blink pelan = mode konfigurasi
      digitalWrite(LED_SETTING, (millis() / 400) % 2);
    }
  }

  // ---- Inisialisasi LoRa ----
  LoRa.setPins(SS, RST, DIO0);
  if (!LoRa.begin(LORA_FREQ)) {
    Serial.println("LoRa gagal init");
    // Blink 2x cepat di LED_SETTING lalu berhenti
    // for (int i = 0; i < 5; i++) {
    //   digitalWrite(LED_SETTING, HIGH);
    //   delay(150);
    //   digitalWrite(LED_SETTING, LOW);
    //   delay(150);
    // }
  } else {
    LoRa.setSpreadingFactor(LORA_SF);
    LoRa.setSignalBandwidth(LORA_BAND);
    LoRa.setTxPower(LORA_TXPOW);
    LoRa.enableCrc();
    loraReady = true;
    Serial.println("LoRa siap");
  }

  // ---- Inisialisasi koneksi & MQTT ----
  bool mqttOK = false;

  SerialSIM.begin(9600, SERIAL_8N1, MODEM_RX, MODEM_TX);
  if (config.koneksi == "wifi") {
    Serial.println("Mode koneksi: WiFi");
    WiFi.begin(config.wifiSSID.c_str(), config.wifiPass.c_str());
    int retry = 0;
    while (WiFi.status() != WL_CONNECTED && retry < 30) {
      delay(300);
      Serial.print(".");
      retry++;
      digitalWrite(LED_RUN, !digitalRead(LED_RUN));  // blink progress
    }
    if (WiFi.status() == WL_CONNECTED) {
      Serial.printf("\nWiFi tersambung: %s\n", WiFi.localIP().toString().c_str());
      static WiFiClient wifiClient;
      mqttClient.setClient(wifiClient);
      mqttClient.setServer(config.mqttHost.c_str(), config.mqttPort);
      if (mqttClient.connect("ESP32WiFi", config.mqttUser.c_str(), config.mqttPass.c_str())) {
        Serial.println("MQTT via WiFi connected!");
        mqttOK = true;
      } else {
        Serial.println("MQTT WiFi gagal connect");
      }
    } else {
      Serial.println("Gagal konek WiFi, fallback ke modem");
      // config.koneksi = "modem";
    }
  }
  if (config.koneksi == "modem") {
    Serial.println("Mode koneksi: Modem SIM800L");
    modem.restart();

    unsigned long lastBlink = 0;
    const unsigned long blinkInterval = 200;

    Serial.println("Menunggu registrasi jaringan seluler...");
    while (!modem.waitForNetwork(1000)) {
      unsigned long now = millis();
      if (now - lastBlink >= blinkInterval) {
        digitalWrite(LED_RUN, !digitalRead(LED_RUN));
        lastBlink = now;
      }
      // biar tidak freeze, kasih yield
      yield();
    }

    Serial.println("\nJaringan seluler OK, mencoba aktifkan GPRS...");
    lastBlink = 0;
    while (!modem.gprsConnect(config.apn.c_str(), config.simUser.c_str(), config.simPass.c_str())) {
      unsigned long now = millis();
      if (now - lastBlink >= 400) { // beda interval
        digitalWrite(LED_RUN, !digitalRead(LED_RUN));
        lastBlink = now;
      }
      yield();
    }

    if (modem.isGprsConnected()) {
      digitalWrite(LED_RUN, LOW);
      Serial.println("\nGPRS aktif, mencoba MQTT...");

      lastBlink = 0;
      mqttClient.setServer(config.mqttHost.c_str(), config.mqttPort);
      while (!mqttClient.connect("ESP32GSM", config.mqttUser.c_str(), config.mqttPass.c_str())) {
        unsigned long now = millis();
        if (now - lastBlink >= 500) {
          digitalWrite(LED_RUN, !digitalRead(LED_RUN));
          lastBlink = now;
        }
        yield();
      }

      digitalWrite(LED_RUN, LOW);
      Serial.println("\nMQTT via GPRS connected!");
      mqttOK = true;
    }
  }

  // if (config.koneksi == "modem") {
  //   Serial.println("Mode koneksi: Modem SIM800L");
  //   modem.restart();
  //   if (modem.waitForNetwork(60000)) {
  //     modem.gprsConnect(config.apn.c_str(), config.simUser.c_str(), config.simPass.c_str());
  //     if (modem.isGprsConnected()) {
  //       mqttClient.setServer(config.mqttHost.c_str(), config.mqttPort);
  //       if (mqttClient.connect("ESP32GSM", config.mqttUser.c_str(), config.mqttPass.c_str())) {
  //         Serial.println("MQTT via GPRS connected!");
  //         mqttOK = true;
  //       } else {
  //         Serial.println("MQTT GPRS gagal connect");
  //       }
  //     } else Serial.println("GPRS gagal aktif");
  //   } else Serial.println("Modem gagal registrasi jaringan");
  // }

  // ---- Indikator Status Final ----
  if (loraReady && mqttOK) {
    digitalWrite(LED_RUN, HIGH);    
    digitalWrite(LED_SETTING, LOW);
  } else {
    //  Ada kegagalan → LED_SETTING blink lambat
    Serial.println("💥 Kegagalan init, restart dalam 5 detik...");
    for (int i = 0; i < 10; i++) {
      digitalWrite(LED_SETTING, !digitalRead(LED_SETTING));
      delay(500);
    }
    ESP.restart();
  }

  Serial.println("\n=== STATUS INISIALISASI ===");
  Serial.printf("LoRa : %s\n", loraReady ? "OK" : "FAIL");
  Serial.printf("MQTT : %s\n", mqttOK ? "OK" : "FAIL");
  Serial.println("============================");
}

void loop() {
  server.handleClient();  

  if (!mqttClient.connected()) mqttReconnect();
  mqttClient.loop();

  if (millis() - lastPing > 30000 + random(0, 5000)) {
    sendPing();
    lastPing = millis();
  }

  // LoRa receive
  int sz = LoRa.parsePacket();
  if (sz) {
    String data = "";
    while (LoRa.available()) data += (char)LoRa.read();
    int rssi = LoRa.packetRssi();
    blinkLED(LED_RX);
    gw_handlePacket(data, rssi);
  }
}
